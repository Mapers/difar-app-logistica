# Banco de pruebas del motor en la PC (no se usa en la app).
#   cmake -S . -B build -DZXING_DIR=/ruta/a/zxing-cpp && cmake --build build && ./build/banco
cmake_minimum_required(VERSION 3.16)
project(banco_motor_lector CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release)
set(BUILD_SHARED_LIBS OFF)
set(ZXING_READERS ON)
set(ZXING_WRITERS OLD CACHE STRING "" FORCE)
set(ZXING_C_API OFF)
set(ZXING_EXAMPLES OFF)
add_subdirectory(${ZXING_DIR} zxing)
add_executable(banco banco.cpp ../src/motor_lector.cpp ../src/preparar.cpp ../src/ean_gris.cpp ../src/red_ean.cpp ../src/red_restaurar.cpp)
target_include_directories(banco PRIVATE ../src)
target_compile_options(banco PRIVATE -Wno-deprecated-declarations)
target_link_libraries(banco ZXing::ZXing)
# Con -DOPENCV_PREFIX=<instalacion de OpenCV estatico con core, imgproc y
# objdetect> se compila con el detector de OpenCV (ML_CON_OPENCV).
if (OPENCV_PREFIX)
  target_sources(banco PRIVATE ../src/detector.cpp)
  target_compile_definitions(banco PRIVATE ML_CON_OPENCV)
  target_include_directories(banco PRIVATE ${OPENCV_PREFIX}/include/opencv4)
  foreach(m objdetect calib3d features2d flann imgproc core)
    target_link_libraries(banco ${OPENCV_PREFIX}/lib/libopencv_${m}.a)
  endforeach()
  find_package(ZLIB)
  target_link_libraries(banco ${CMAKE_DL_LIBS} pthread $<$<BOOL:${ZLIB_FOUND}>:ZLIB::ZLIB>)
  message(STATUS "banco con OpenCV de ${OPENCV_PREFIX}")
endif()
